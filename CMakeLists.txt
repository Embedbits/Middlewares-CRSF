# Same principle as ModBus: the first time the project is configured, the
# example is copied into the project handler module folder Middlewares/CRSF
# (Crsf_Example.cpp, Crsf_Example.h, Crsf_Hal.h, Crsf_Phy_Usart.*,

#========================== Handler module seeding ============================#

# Middlewares/Core/CRSF -> Middlewares
get_filename_component(CRSF_CORE_DIR "${CMAKE_CURRENT_LIST_DIR}" DIRECTORY)
get_filename_component(CRSF_CORE_DIR_NAME "${CRSF_CORE_DIR}" NAME)
get_filename_component(CRSF_MIDDLEWARES_DIR "${CRSF_CORE_DIR}" DIRECTORY)

set(CRSF_EXAMPLE_DIR  "${CMAKE_CURRENT_LIST_DIR}/Example")
set(CRSF_HANDLER_DIR  "${CRSF_MIDDLEWARES_DIR}/CRSF")

#------------------------------------------------------------------------------#
# Copies the example into project handler module folder Middlewares/CRSF if
# the folder does not exist, it is empty or it contains only the unmodified
# module scaffold generated by EmBi platform (Setup - Configure middleware
# module, ModuleInit - source with empty <Name>_Task). Otherwise the folder is
# the project's own handler module and it is left untouched.
#
# Middlewares.cmake processes Core/ first, so the handler module folder is
# already seeded when Middlewares.cmake adds it to the build.
#------------------------------------------------------------------------------#
function(Crsf_SeedHandlerModule)

    # Component is not vendored in a project (eg. standalone unit testing)
    if(NOT CRSF_CORE_DIR_NAME STREQUAL "Core")
        message(STATUS "CRSF: not located in Middlewares/Core, handler module is not created.")
        return()
    endif()

    file(GLOB HANDLER_SOURCES "${CRSF_HANDLER_DIR}/*.c" "${CRSF_HANDLER_DIR}/*.cpp")

    set(SCAFFOLD_SOURCE "")

    if(HANDLER_SOURCES)

        list(LENGTH HANDLER_SOURCES HANDLER_SOURCES_COUNT)

        if(HANDLER_SOURCES_COUNT EQUAL 1)
            file(READ "${HANDLER_SOURCES}" SCAFFOLD_CONTENT)

            # Unmodified scaffold of EmBi ModuleInit - empty task function
            if(SCAFFOLD_CONTENT MATCHES "This function shall be called in the main loop of the application" AND
               SCAFFOLD_CONTENT MATCHES "void [A-Za-z0-9_]+_Task\\( void \\)[\r\n]+\\{[ \t\r\n]*\\}")
                set(SCAFFOLD_SOURCE "${HANDLER_SOURCES}")
            endif()
        endif()

        if(NOT SCAFFOLD_SOURCE)
            message(STATUS "CRSF: handler module ${CRSF_HANDLER_DIR} found.")
            return()
        endif()

    endif()

    file(MAKE_DIRECTORY "${CRSF_HANDLER_DIR}")

    file(GLOB EXAMPLE_FILES LIST_DIRECTORIES false "${CRSF_EXAMPLE_DIR}/*")

    foreach(EXAMPLE_FILE IN LISTS EXAMPLE_FILES)
        get_filename_component(EXAMPLE_FILE_NAME "${EXAMPLE_FILE}" NAME)
        file(COPY_FILE "${EXAMPLE_FILE}" "${CRSF_HANDLER_DIR}/${EXAMPLE_FILE_NAME}")
    endforeach()

    # Scaffold (source and its headers) is replaced by the example
    if(SCAFFOLD_SOURCE)
        get_filename_component(SCAFFOLD_NAME "${SCAFFOLD_SOURCE}" NAME_WE)
        file(REMOVE "${SCAFFOLD_SOURCE}"
                    "${CRSF_HANDLER_DIR}/${SCAFFOLD_NAME}.h"
                    "${CRSF_HANDLER_DIR}/${SCAFFOLD_NAME}_Port.h"
                    "${CRSF_HANDLER_DIR}/${SCAFFOLD_NAME}_Types.h")
    endif()

    message(STATUS "CRSF: handler module created from example in ${CRSF_HANDLER_DIR} - "
                   "implement Crsf_Hal.h and adapt it to the application.")

endfunction()


Crsf_SeedHandlerModule()

#============================ Stack library ===================================#

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/CRSF)
